import json, sys, time
sys.argv = [sys.argv[0]] + sys.argv[1:]
src = open('measure_engine.py').read().rsplit('main(sys.argv[1])', 1)[0]
exec(src)

def case(name, fn):
    path, evalv, evaln = ENGINES[name]
    log = []
    e, r, names = boot(path, evalv, evaln, log)
    try:
        res = fn(e, names)
    finally:
        if e.p.poll() is None:
            e.p.kill()
            e.p.wait()
    return res, log

def go_form(cmd):
    def f(e, names):
        return go_case(e, cmd, 6)
    return f

def mate(e, names):
    # 1手詰（持ち駒の金を打つ）
    e.send("position sfen 8k/9/8P/9/9/9/9/9/9 b G2r2b3g4s4n4l17p 1")
    e.send("go mate 2000")
    t, l, seen = e.wait(lambda s: s.startswith(("checkmate", "bestmove")), 5)
    return {"cmd": "go mate 2000", "after_ms": ms(t), "line": l}

def overlap(e, names):
    e.send("position startpos"); e.send("go infinite"); e.drain(0.5)
    e.send("position startpos moves 7g7f"); e.send("go infinite")
    seen = e.drain(1.0)
    e.send("stop")
    t, l, s2 = e.wait(lambda x: x.startswith("bestmove"), 5)
    second = None
    if l:
        e.send("stop")
        t3, l3, _ = e.wait(lambda x: x.startswith("bestmove"), 3)
        second = l3
    return {"bestmove_during_overlap": [s for s in seen if s.startswith("bestmove")], "stop_then_bestmove_ms": ms(t), "line": l, "second_stop": second, "alive": e.p.poll() is None}

def quit_searching(e, names):
    e.send("position startpos"); e.send("go infinite"); e.drain(0.5)
    s = time.monotonic(); e.send("quit")
    try:
        e.p.wait(timeout=8); return {"quit_ms": ms(time.monotonic()-s), "code": e.p.returncode}
    except Exception:
        return {"quit_ms": None}

def stop_lat(e, names):
    out = []
    for _ in range(3):
        e.send("position startpos"); e.send("go infinite"); e.drain(1.0)
        e.send("stop"); t, l, _ = e.wait(lambda x: x.startswith("bestmove"), 5); out.append(ms(t))
    return out

def multipv_live(e, names):
    if "MultiPV" not in names: return "not declared"
    e.send("position startpos"); e.send("go infinite"); e.drain(0.8)
    e.send("setoption name MultiPV value 3")
    seen = e.drain(2.0)
    mp = sorted({s.split(" multipv ")[1].split()[0] for s in seen if " multipv " in s})
    e.send("stop"); e.wait(lambda x: x.startswith("bestmove"), 5)
    e.send("position startpos"); e.send("go infinite")
    seen = e.drain(1.0)
    mp2 = sorted({s.split(" multipv ")[1].split()[0] for s in seen if " multipv " in s})
    return {"during_same_search": mp, "next_go": mp2}

def bad_values(e, names):
    r = {}
    e.send("setoption name NoSuchOption value 1"); r["unknown"] = e.drain(0.4)
    e.send("setoption name USI_Hash value 99999999999"); r["hash_out_of_range"] = e.drain(0.4)
    e.send("isready"); t, l, seen = e.wait(lambda s: s == "readyok", 20)
    r["isready_after"] = {"ms": ms(t), "extra": [s for s in seen if s != "readyok"][:4]}
    r["alive"] = e.p.poll() is None
    return r

def bad_eval(name):
    path, _, evaln = ENGINES[name]
    log = []
    e = Engine(path, log); e.send("usi"); e.wait(lambda s: s == "usiok", 10)
    e.send(f"setoption name {evaln} value /nonexistent/path")
    e.send("isready")
    t, l, seen = e.wait(lambda s: s == "readyok", 10)
    res = {"readyok_ms": ms(t), "line": l, "lines": seen[:6], "alive": e.p.poll() is None, "code": e.p.poll()}
    if e.p.poll() is None:
        e.p.kill(); e.p.wait()
    return res

name = sys.argv[1]
out = {"engine": name}
biglog = []
for label, fn in [("stop_latency", stop_lat), ("go_movetime", go_form("go movetime 1000")), ("go_depth", go_form("go depth 5")), ("go_nodes", go_form("go nodes 20000")),
                  ("go_byoyomi", go_form("go byoyomi 1000")), ("go_btime0_byoyomi", go_form("go btime 0 wtime 0 byoyomi 1000")), ("go_mate", mate),
                  ("go_while_searching", overlap), ("quit_while_searching", quit_searching), ("multipv_live", multipv_live), ("bad_values", bad_values)]:
    res, log = case(name, fn)
    out[label] = res
    biglog += [f"=== {label}"] + log
out["bad_eval"] = bad_eval(name)
json.dump(out, open(f"{name}.2.json", "w"), ensure_ascii=False, indent=1)
open(f"{name}.2.log", "w").write("\n".join(biglog) + "\n")
